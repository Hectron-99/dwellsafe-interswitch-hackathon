async function redirectUser(userId) { const { data: profile, error } = await supabaseClient .from('profiles') .select('role') .eq('id', userId) .single(); // Get the current path base (handles GitHub Pages repo structure automatically) const baseFolder = window.location.pathname.substring(0, window.location.pathname.lastIndexOf('/') + 1); if (error || !profile) { window.location.href = baseFolder + 'student.html'; return; } if (profile.role === 'admin') { window.location.href = baseFolder + 'admin.html'; } else if (profile.role === 'landlord') { window.location.href = baseFolder + 'landlord.html'; } else { window.location.href = baseFolder + 'student.html'; } }
